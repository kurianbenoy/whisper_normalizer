Normalizer

def Normalizer(
    language:str, **options
):

Normalize text for one language: Normalizer("hi", tts_mode=True)("...").

language accepts ISO codes ("hi"), BCP-47 tags ("hi-IN", "zh-Hans") and English names ("hindi"). Options such as tts_mode=True are validated against the selected language and raise ValueError if it does not accept them, instead of being ignored. Normalizer.for_language("hi") is an equivalent spelling.

source

LanguageSupport

def LanguageSupport(
    code:str, name:str, factory:Callable[..., Callable[[str], str]], capabilities:tuple[str, ...], tier:str,
    marks:Literal['kept', 'stripped']='kept'
)->None:

A supported language and the normalization level available for it.

tier is one of:

  • "dedicated": a normalizer written and reviewed for this language.
  • "script": no dedicated class yet, but the language shares a script with a dedicated normalizer that is reused as-is (for example Marathi reuses the Devanagari normalizer, without Hindi’s number/TTS tuning).
  • "generic": only conservative, script-safe cleanup is applied; no language-specific rules exist.

Normalizer: the single public entry point

Normalizer("hi") and Normalizer.for_language("hi") are equivalent. The tests below use only that public interface; expected values are literals.

# Normalizer(language) selects the normalizer for a language and is callable. One case per tier.
CASES = [
    ("hi", "नमस्ते, दुनिया!", "नमस्ते, दुनिया!"),  # dedicated, Indic
    ("en", "Colour and favour.", "color and favor"),  # dedicated, English
    ("fr", "Vingt et un euros.", "21 euros"),  # dedicated, international
    ("zh", "我有二十一個蘋果。", "我有21個蘋果"),  # dedicated, non-Latin
    ("es", "El total es 1.234,50.", "el total es 1234.50"),  # locale numbers
    ("de", "Guten Tag!", "guten tag"),  # generic tier
]
bad = {}
for code, text, expected in CASES:
    out = Normalizer(code)(text)
    if out != expected:
        bad[code] = (out, expected)
assert not bad, bad
# Normalizer.for_language(language) is the same thing, spelled as a constructor method.
bad = {}
for code, text, expected in CASES:
    n = Normalizer.for_language(code)
    if not isinstance(n, Normalizer) or n(text) != expected:
        bad[code] = n(text)
assert not bad, bad
# Metadata: the language is resolved once, from a code, a BCP-47 tag or an English name.
for query, code, name, tier in [
    ("hi", "hi", "Hindi", "dedicated"),
    ("hi-IN", "hi", "Hindi", "dedicated"),
    ("Hindi", "hi", "Hindi", "dedicated"),
    ("zh-Hans", "zh", "Chinese", "dedicated"),
    ("odia", "or", "Odia", "dedicated"),
    ("mr", "mr", "Marathi", "script"),
    ("de", "de", "German", "generic"),
]:
    n = Normalizer(query)
    assert (n.language, n.name, n.tier) == (code, name, tier), (query, n.language, n.name, n.tier)

assert "tts_mode" in Normalizer("hi").capabilities

Capability contract

The registry’s capability metadata is checked against real behaviour for every language, so the documented table cannot drift from what the normalizers do.

# tts_mode is advertised exactly when it can be constructed, and then it spells digits as words.
bad = {}
for entry in supported_languages():
    try:
        n = Normalizer(entry.code, tts_mode=True)
    except ValueError:
        constructible = False
    else:
        constructible = True
        spoken = n("5")
        if any(ch.isdigit() for ch in spoken):
            bad[entry.code] = f"tts_mode left a digit in {spoken!r}"
    if constructible != ("tts_mode" in entry.capabilities):
        bad[entry.code] = f"advertises tts_mode={'tts_mode' in entry.capabilities} but constructible={constructible}"
assert not bad, bad
# digits_to_words says when digits become words, and matches what default mode really does for all languages:
# "always" (Indic family), "tts_mode" (only with tts_mode=True) or "never" (tts_mode unavailable).
bad = {}
for entry in supported_languages():
    spelled_by_default = not any(ch.isdigit() for ch in Normalizer(entry.code)("5"))
    if (entry.digits_to_words == "always") != spelled_by_default:
        bad[entry.code] = (entry.digits_to_words, "spelled by default" if spelled_by_default else "digits kept")
    # Outside the Indic family, "never" is exactly "tts_mode unavailable".
    if entry.digits_to_words != "always" and (entry.digits_to_words == "never") != ("tts_mode" not in entry.capabilities):
        bad[entry.code] = (entry.digits_to_words, entry.capabilities)
assert not bad, bad
assert {e.code for e in supported_languages() if e.digits_to_words == "always"} == {
    "as", "bn", "gu", "hi", "kn", "ml", "mr", "ne", "or", "pa", "sa", "ta", "te"}
assert Normalizer("en").tts_mode is False and [e for e in supported_languages() if e.code == "en"][0].digits_to_words == "tts_mode"
# marks says whether combining marks survive normalization ("kept") or are removed ("stripped"); it is checked against
# real behaviour for every dedicated and script-tier language, plus the generic tier.
import unicodedata


def count_marks(s):
    return sum(unicodedata.category(ch).startswith("M") for ch in unicodedata.normalize("NFD", s))


INDIC_BASE = {"hi": 0x0900, "mr": 0x0900, "ne": 0x0900, "sa": 0x0900, "bn": 0x0980, "as": 0x0980, "pa": 0x0A00,
              "gu": 0x0A80, "or": 0x0B00, "ta": 0x0B80, "te": 0x0C00, "kn": 0x0C80, "ml": 0x0D00}
MARK_SAMPLES = {code: chr(b + 0x15) + chr(b + 0x3F) for code, b in INDIC_BASE.items()}  # KA + vowel sign I
MARK_SAMPLES.update({"en": "café", "fr": "café", "es": "niño", "ar": "أَهْلًا", "ru": "йод", "de": "café", "th": "ที่"})
NO_MARKS_IN_SCRIPT = {"zh"}  # Chinese has no combining marks to keep or strip

bad = {}
checked_marks = set()
for entry in supported_languages():
    # Every generic-tier language runs the same script-safe cleanup, so one accented sample covers each of them.
    sample = MARK_SAMPLES.get(entry.code) or ("café" if entry.tier == "generic" else None)
    if sample is None:
        assert entry.code in NO_MARKS_IN_SCRIPT, f"no marks sample for {entry.code}"
        continue
    checked_marks.add(entry.code)
    out = Normalizer(entry.code)(sample)
    expected_marks = 0 if entry.marks == "stripped" else count_marks(sample)
    if count_marks(out) != expected_marks:
        bad[entry.code] = (entry.marks, sample, out)
assert not bad, bad
assert {e.code for e in supported_languages() if e.marks == "stripped"} == {"ar", "en"}
# Every language except Chinese (no combining marks) is checked: 100 of 101.
assert len(checked_marks) == 100 and checked_marks == {e.code for e in supported_languages()} - NO_MARKS_IN_SCRIPT

Capability table

The table in README.md and nbs/index.ipynb is generated from the registry by capability_section() below; the test pins both files to it, so it cannot go stale.

import json
from pathlib import Path

from indic_numtowords import supported_langs as _numtowords_langs


def capability_section():
    """Markdown for the supported-languages table: one row per dedicated/script language, plus a generic-tier summary."""
    rows = ["| Language | Code | Normalizer | Tier | Digits → words | Marks | `tts_mode` |", "|---|---|---|---|---|---|---|"]
    entries = sorted((e for e in supported_languages() if e.tier != "generic"), key=lambda e: (e.tier != "dedicated", e.name))
    for e in entries:
        cls = getattr(e.factory, "func", e.factory).__name__
        digits = e.digits_to_words
        if digits == "always" and e.code not in _numtowords_langs:
            digits += " (Hindi number words)"
        tts = "yes" if "tts_mode" in e.capabilities else "no"
        rows.append(f"| {e.name} | `{e.code}` | `{cls}` | {e.tier} | {digits} | {e.marks} | {tts} |")
    generic = [e for e in supported_languages() if e.tier == "generic"]
    with_tts = sum("tts_mode" in e.capabilities for e in generic)
    summary = (
        f"The other {len(generic)} languages use the generic tier: Unicode NFC, brackets and punctuation removed, "
        f"case folded, combining marks kept. Digits become words only with `tts_mode=True`, which {with_tts} of them support."
    )
    return "\n".join(rows) + "\n\n" + summary
