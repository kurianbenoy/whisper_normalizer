NormalizerI

def NormalizerI(
    *args, **kwargs
):

The normalizer classes do the following: * Some characters have multiple Unicode codepoints. The normalizer chooses a single standard representation * Some control characters are deleted * While typing using the Latin keyboard, certain typical mistakes occur which are corrected by the module Base class for normalizer. Performs some common normalization, which includes: * Byte order mark, word joiner, etc. removal * ZERO_WIDTH_NON_JOINER and ZERO_WIDTH_JOINER removal * ZERO_WIDTH_SPACE and NO_BREAK_SPACE replaced by spaces Script specific normalizers should derive from this class and override the normalize() method. They can call the super class ’normalize() method to avail of the common normalization

    return num2words(str(number), lang=code)

# A colon after a character of the script becomes that script's visarga; the character itself is kept.
# Unicode block bases are fixed by the standard: consonant KA = base+0x15, vowel sign I = base+0x3F, visarga = base+0x03.
BLOCK_BASE = {"hi": 0x0900, "bn": 0x0980, "pa": 0x0A00, "gu": 0x0A80, "or": 0x0B00,
              "ta": 0x0B80, "te": 0x0C00, "kn": 0x0C80, "ml": 0x0D00}
bad = {}
for code, spec in INDIC_TTS.items():
    base = BLOCK_BASE[code]
    word = chr(base + 0x15) + chr(base + 0x3F)
    out = spec["cls"]()(word + ":")
    expected = word + chr(base + 0x03)
    if out != expected:
        bad[code] = (out, expected)
assert not bad, bad
# A colon that does not follow a character of the script is left alone.

def mismatches(make_case):
    """Run make_case(code, spec) -> (got_thunk, expected) for every language; return {code: got-or-exception} for failures."""
    bad = {}
    for code, spec in INDIC_TTS.items():
        got, expected = make_case(code, spec)
        try:
            value = got()
        except Exception as e:  # a crash is a failure to report, not to hide
            bad[code] = f"raised {type(e).__name__}: {e}"
            continue
        if value != expected:
            bad[code] = (value, expected)
    return bad
# Digits are spelled as words, and each number is replaced on its own: a "1" must not rewrite the "1" inside "10".
bad = mismatches(lambda code, spec: (lambda: spec["cls"]()("1 10"), f"{spoken(code, 1)} {spoken(code, 10)}"))
assert not bad, bad
# Default mode: a decimal number must not crash; each side of the dot is spelled and the dot is kept.
bad = mismatches(lambda code, spec: (lambda: spec["cls"]()("1.5"), f"{spoken(code, 1)}.{spoken(code, 5)}"))
assert not bad, bad
# tts_mode: a decimal is spoken as <whole> <point-word> <fraction>.
bad = mismatches(lambda code, spec: (lambda: spec["cls"](tts_mode=True)("1.5"), f"{spoken(code, 1)} {POINT[code]} {spoken(code, 5)}"))
assert not bad, bad


# Multi-digit fraction (existing behaviour, pinned): Hindi reads the fraction digit by digit; the other languages read it as one number.
def fraction_expected(code):
    frac = f"{spoken(code, 2)} {spoken(code, 5)}" if code == "hi" else spoken(code, 25)
    return f"{spoken(code, 1)} {POINT[code]} {frac}"


bad = mismatches(lambda code, spec: (lambda: spec["cls"](tts_mode=True)("1.25"), fraction_expected(code)))
assert not bad, bad
